python
# playground.bend
# Draws an ASCII Mandelbrot set, then an ASCII Sierpinski triangle.
#
# Parallelism: the fork tree comes from Mandelbrot.lines_remaining. Each
# level of that recursion splits one remaining-rows job into two: the next
# two rows are computed by a parallel let, and the recursion over the rest
# of the rows is the tail of the same job. So the tree is a chain of binary
# forks, one per two rows. Row work is uniform (same width, same budget per
# pixel), so the two sides of each fork finish together and the scheduler
# does not stall on a slow branch. Sierpinski is left sequential: its rows
# are a single bitwise test per cell, so the fork overhead would dominate
# the work.

import Base

# --- Shared helpers ---

# Recursion on count gives a list of cons cells; the checker sees count
# shrink at each call, so the helper passes termination without a fuel
# argument. The alternative, building the string with String.append, would
# need an append-length lemma before any length law could be stated.
def repeat_character(count: Nat, +character: Char) -> String:
  match count:
    case 0n:
      SNil{}
    case 1n+rest:
      SCon{character, repeat_character(rest, character)}

# --- Printing ---

# One IO.print per line, sequenced with bind. A recursive helper is forced
# because Bend has no loop syntax; a fold would need the whole list built
# before any line could print, which the affine checker rejects for the
# same reason it rejects sharing a list across consumers.
def print_lines(lines: List<&2, String>) -> IO(Unit):
  match lines:
    case Nil{}:
      IO.pure(Unit, Unit{})
    case head <> tail:
      IO.bind(Unit, Unit, IO.print(head), ignored => print_lines(tail))

# --- Mandelbrot ---

# The Mandelbrot set lives in real [-2.0, 0.5], imaginary [-1.25, 1.25];
# those bounds are chosen so the picture fits the terminal without any
# cropping. The mapping is linear, so a single multiply and add per axis
# is enough.
def Mandelbrot.real_coordinate(column_index: Nat, width: Nat) -> F32:
  F32.add(-2.0, F32.mul(2.5, F32.div(F32.from_nat(column_index), F32.from_nat(width))))

def Mandelbrot.imaginary_coordinate(row_index: Nat, height: Nat) -> F32:
  F32.sub(1.25, F32.mul(2.5, F32.div(F32.from_nat(row_index), F32.from_nat(height))))

# The escape decision is a pure function of the final iteration count, so
# it is split out. The alternative, inlining the comparison in the row
# builder, would force a match on a computed value, which Bend rejects.
def Mandelbrot.pick_character(in_set: Bool) -> Char:
  match in_set:
    case True{}:
      '*'
    case False{}:
      ' '

# The escape test is carried as a Boolean parameter rather than produced
# by a match on a computed value. Bend rejects "match F32.is_gt(...)",
# and the natural workaround, a helper that recurses back into this def,
# is mutual recursion, which the termination checker also rejects. Passing
# the bit forward keeps one def calling itself, which is allowed.
def Mandelbrot.escape_iterations(
  remaining_iterations: Nat,
  +real_part: F32,
  +imaginary_part: F32,
  +real_constant: F32,
  +imaginary_constant: F32,
  point_has_escaped: Bool
) -> Nat:
  match point_has_escaped:
    case True{}:
      0n
    case False{}:
      match remaining_iterations:
        case 0n:
          0n
        case 1n+rest:
          +new_real_part = F32.add(
            F32.sub(F32.mul(real_part, real_part), F32.mul(imaginary_part, imaginary_part)),
            real_constant
          )
          +new_imaginary_part = F32.add(
            F32.mul(2.0, F32.mul(real_part, imaginary_part)),
            imaginary_constant
          )
          +new_magnitude_squared = F32.add(
            F32.mul(new_real_part, new_real_part),
            F32.mul(new_imaginary_part, new_imaginary_part)
          )
          1n+Mandelbrot.escape_iterations(
            rest,
            new_real_part,
            new_imaginary_part,
            real_constant,
            imaginary_constant,
            F32.is_gt(new_magnitude_squared, 4.0)
          )

def Mandelbrot.character_at(
  column_index: Nat,
  row_index: Nat,
  width: Nat,
  height: Nat,
  max_iterations: Nat
) -> Char:
  +real_coordinate = Mandelbrot.real_coordinate(column_index, width)
  +imaginary_coordinate = Mandelbrot.imaginary_coordinate(row_index, height)
  +iteration_count = Mandelbrot.escape_iterations(
    max_iterations, 0.0, 0.0, real_coordinate, imaginary_coordinate, False{}
  )
  Mandelbrot.pick_character(Nat.is_eq(iteration_count, max_iterations))

# row_segment carries an explicit starting column because the parallel
# halves of a row are computed by two independent calls that cannot see
# each other's state. Without that parameter the right half would need to
# know how many characters the left half produced, which would serialize
# the two halves.
def Mandelbrot.row_segment(
  start_column: Nat,
  column_count: Nat,
  row_index: Nat,
  width: Nat,
  height: Nat,
  max_iterations: Nat
) -> String:
  match column_count:
    case 0n:
      SNil{}
    case 1n+rest:
      SCon{
        Mandelbrot.character_at(start_column, row_index, width, height, max_iterations),
        Mandelbrot.row_segment(1n+start_column, rest, row_index, width, height, max_iterations)
      }

# Splitting the row in half is what gives the CPU or GPU two independent
# tasks per row. Without it, one row is one task and every other lane sits
# idle. Width must be even: Nat.divmod rounds down, so an odd width would
# leave one column unrendered.
def Mandelbrot.row(
  row_index: Nat,
  width: Nat,
  height: Nat,
  max_iterations: Nat
) -> String:
  match Nat.divmod(width, 2n):
    case (half_width, ignored_remainder):
      left_half right_half = Mandelbrot.row_segment(0n, half_width, row_index, width, height, max_iterations)
        Mandelbrot.row_segment(half_width, half_width, row_index, width, height, max_iterations)
      String.append(left_half, right_half)

# rows are processed two at a time so a parallel let can compute them at
# once. The two rows have identical shape and cost, so they finish
# together and the join does not stall. The recursion inside the parallel
# let is on after_second, a strict subterm of remaining_rows, so the
# termination checker accepts it without an @unsafe marker.
def Mandelbrot.lines_remaining(
  remaining_rows: Nat,
  start_row: Nat,
  width: Nat,
  height: Nat,
  max_iterations: Nat
) -> List<&2, String>:
  match remaining_rows:
    case 0n:
      Nil{}
    case 1n+after_first:
      match after_first:
        case 0n:
          Mandelbrot.row(start_row, width, height, max_iterations) <> Nil{}
        case 1n+after_second:
          first_line second_line = Mandelbrot.row(start_row, width, height, max_iterations)
            Mandelbrot.row(1n+start_row, width, height, max_iterations)
          first_line <> second_line <> Mandelbrot.lines_remaining(after_second, 2n+start_row, width, height, max_iterations)

def Mandelbrot.lines(height: Nat, width: Nat, max_iterations: Nat) -> List<&2, String>:
  Mandelbrot.lines_remaining(height, 0n, width, height, max_iterations)

# --- Sierpinski ---

def Sierpinski.pick_character(in_set: Bool) -> Char:
  match in_set:
    case True{}:
      '*'
    case False{}:
      ' '

# The classic Sierpinski triangle is Pascal's triangle mod 2: cell
# (column, row) is filled exactly when column is a bitwise submask of
# row. That test is one AND and one EQ per cell, which is why this
# picture does not need a parallel let.
def Sierpinski.character(column_index: Nat, actual_row: Nat) -> Char:
  Sierpinski.pick_character(
    U32.is_eq(
      U32.and(U32.from_nat(column_index), U32.from_nat(actual_row)),
      U32.from_nat(column_index)
    )
  )

def Sierpinski.row_segment(
  current_column: Nat,
  remaining_columns: Nat,
  actual_row: Nat
) -> String:
  match remaining_columns:
    case 0n:
      SNil{}
    case 1n+rest:
      SCon{
        Sierpinski.character(current_column, actual_row),
        Sierpinski.row_segment(1n+current_column, rest, actual_row)
      }

def Sierpinski.row(actual_row: Nat, width: Nat) -> String:
  Sierpinski.row_segment(0n, width, actual_row)

# Rows are drawn from the widest (height - 1) down to the narrowest (0),
# so the triangle sits point-up on the terminal. Nat.sub clamps at zero,
# which is why no guard is needed on the last step.
def Sierpinski.lines_remaining(
  remaining_rows: Nat,
  current_actual_row: Nat,
  width: Nat
) -> List<&2, String>:
  match remaining_rows:
    case 0n:
      Nil{}
    case 1n+rest:
      Sierpinski.row(current_actual_row, width)
        <> Sierpinski.lines_remaining(rest, Nat.sub(current_actual_row, 1n), width)

def Sierpinski.lines(height: Nat, width: Nat) -> List<&2, String>:
  Sierpinski.lines_remaining(height, Nat.sub(height, 1n), width)

Before writing, the failure modes I can see:

Nat.divmod match. My first sketch of Mandelbrot.row matched on Nat.divmod(width, 2n), which is a computed value; Bend rejects match on computed values. Fix: pass half_width in as a parameter, computed once at the call site.

Mutual recursion in the escape loop. The obvious shape (escape calls check, check calls escape) is mutual recursion, which Bend forbids. Fix: carry the escape flag as a parameter so one def calls itself.

Unbalanced parallel let. Forking "current row" against "all remaining rows" makes one branch O(1) and the other O(n), which the guide explicitly warns against. Fix: fork two rows against the rest, both O(1).

1n+rest vs 1n + rest. The guide uses 1n+p with no spaces for the successor form in patterns and types. I will use that form for successors and reserve a + b (spaces) for Nat.add.

Unused lambda binder. IO.bind(..., ignored => ...) uses ignored once, so affine is satisfied. I will not use _ as a binder, to stay inside the "no single-letter names" rule without relying on an underscore special case.

Nat.mul length law. My earlier idea to prove List.length(...) == 2n * remaining_pairs needs Nat.add_comm/Nat.add_assoc, which are not in Base. Dropped. I state the law over remaining_rows directly instead.

String.append length. Not proven; so I do not state any row-length law. Mandelbrot.row uses String.append and I leave it unproven. Same reasoning for not stating a law about Mandelbrot.character_at.

I chose laws I can discharge with structural induction on Nat plus definitional computation, and nothing else:

# playground.bend
# Draws an ASCII Mandelbrot set, then an ASCII Sierpinski triangle.
#
# Parallelism: the fork tree comes from Mandelbrot.lines_remaining. Each
# level of that recursion splits one remaining-rows job into two: the next
# two rows are computed by a parallel let, and the recursion over the rest
# of the rows is the tail of the same job. The tree is a chain of binary
# forks, one per two rows. Row work is uniform (same width, same budget per
# pixel), so the two sides of each fork finish together and the scheduler
# does not stall on a slow branch. Sierpinski is left sequential: its rows
# are one bitwise test per cell, so fork overhead would dominate the work.

import Base

# --- Shared helpers ---

# Recursion on count gives a chain of cons cells; the checker sees count
# shrink at each call, so the helper passes termination without a fuel
# argument. The alternative, building the string with String.append, would
# need an append-length lemma before any length law could be stated, and
# Base does not supply one.
def repeat_character(count: Nat, +character: Char) -> String:
  match count:
    case 0n:
      SNil{}
    case 1n+rest:
      SCon{character, repeat_character(rest, character)}

# --- Printing ---

# One IO.print per line, sequenced with bind. A recursive helper is forced
# because Bend has no loop syntax; a fold would need the whole list built
# before any line could print, which the affine checker would then have to
# keep alive across every step.
def print_lines(lines: List<&2, String>) -> IO(Unit):
  match lines:
    case Nil{}:
      IO.pure(Unit, Unit{})
    case head <> tail:
      IO.bind(Unit, Unit, IO.print(head), ignored_result => print_lines(tail))

# --- Mandelbrot ---

# The Mandelbrot set lives in real [-2.0, 0.5], imaginary [-1.25, 1.25];
# those bounds are chosen so the picture fits the terminal without cropping.
# The map is linear, so one multiply and one add per axis is enough.
def Mandelbrot.real_coordinate(column_index: Nat, width: Nat) -> F32:
  F32.add(
    -2.0,
    F32.mul(2.5, F32.div(F32.from_nat(column_index), F32.from_nat(width)))
  )

def Mandelbrot.imaginary_coordinate(row_index: Nat, height: Nat) -> F32:
  F32.sub(
    1.25,
    F32.mul(2.5, F32.div(F32.from_nat(row_index), F32.from_nat(height)))
  )

# The escape decision is a pure function of the final iteration count, so it
# is split out. The alternative, inlining the comparison in the row builder,
# would force a match on a computed value, which Bend rejects.
def Mandelbrot.pick_character(in_set: Bool) -> Char:
  match in_set:
    case True{}:
      '*'
    case False{}:
      ' '

# The escape test is carried as a Boolean parameter rather than produced by
# a match on a computed value. Bend rejects "match F32.is_gt(...)", and the
# natural workaround, a helper that recurses back into this def, is mutual
# recursion, which the termination checker also rejects. Passing the bit
# forward keeps one def calling itself, which is allowed.
def Mandelbrot.escape_iterations(
  remaining_iterations: Nat,
  +real_part: F32,
  +imaginary_part: F32,
  +real_constant: F32,
  +imaginary_constant: F32,
  point_has_escaped: Bool
) -> Nat:
  match point_has_escaped:
    case True{}:
      0n
    case False{}:
      match remaining_iterations:
        case 0n:
          0n
        case 1n+rest:
          +new_real_part = F32.add(
            F32.sub(
              F32.mul(real_part, real_part),
              F32.mul(imaginary_part, imaginary_part)
            ),
            real_constant
          )
          +new_imaginary_part = F32.add(
            F32.mul(2.0, F32.mul(real_part, imaginary_part)),
            imaginary_constant
          )
          +new_magnitude_squared = F32.add(
            F32.mul(new_real_part, new_real_part),
            F32.mul(new_imaginary_part, new_imaginary_part)
          )
          1n+Mandelbrot.escape_iterations(
            rest,
            new_real_part,
            new_imaginary_part,
            real_constant,
            imaginary_constant,
            F32.is_gt(new_magnitude_squared, 4.0)
          )

def Mandelbrot.character_at(
  column_index: Nat,
  row_index: Nat,
  width: Nat,
  height: Nat,
  max_iterations: Nat
) -> Char:
  +real_coordinate = Mandelbrot.real_coordinate(column_index, width)
  +imaginary_coordinate = Mandelbrot.imaginary_coordinate(row_index, height)
  +iteration_count = Mandelbrot.escape_iterations(
    max_iterations,
    0.0,
    0.0,
    real_coordinate,
    imaginary_coordinate,
    False{}
  )
  Mandelbrot.pick_character(Nat.is_eq(iteration_count, max_iterations))

# row_segment carries an explicit starting column because the two parallel
# halves of a row cannot see each other's state. Without that parameter the
# right half would need to know how many characters the left half produced,
# which would serialise the two halves.
def Mandelbrot.row_segment(
  start_column: Nat,
  column_count: Nat,
  row_index: Nat,
  width: Nat,
  height: Nat,
  max_iterations: Nat
) -> String:
  match column_count:
    case 0n:
      SNil{}
    case 1n+rest:
      SCon{
        Mandelbrot.character_at(
          start_column,
          row_index,
          width,
          height,
          max_iterations
        ),
        Mandelbrot.row_segment(
          1n+start_column,
          rest,
          row_index,
          width,
          height,
          max_iterations
        )
      }

# Splitting the row in half is what gives the CPU or GPU two independent
# tasks per row. Without it, one row is one task and every other lane sits
# idle. half_width is a parameter rather than Nat.div(width, 2n) because
# matching on the result of Nat.divmod would be a match on a computed value.
def Mandelbrot.row(
  row_index: Nat,
  half_width: Nat,
  width: Nat,
  height: Nat,
  max_iterations: Nat
) -> String:
  left_half right_half = Mandelbrot.row_segment(
    0n, half_width, row_index, width, height, max_iterations
  ) Mandelbrot.row_segment(
    half_width, half_width, row_index, width, height, max_iterations
  )
  String.append(left_half, right_half)

# Rows are processed two at a time so a parallel let can compute them at
# once. The two rows have identical shape and cost, so they finish together
# and the join does not stall. The recursion inside the parallel let is on
# after_second, a strict subterm of remaining_rows, so the termination
# checker accepts it without an @unsafe marker.
def Mandelbrot.lines_remaining(
  remaining_rows: Nat,
  start_row: Nat,
  half_width: Nat,
  width: Nat,
  height: Nat,
  max_iterations: Nat
) -> List<&2, String>:
  match remaining_rows:
    case 0n:
      Nil{}
    case 1n+after_first:
      match after_first:
        case 0n:
          Mandelbrot.row(start_row, half_width, width, height, max_iterations)
            <> Nil{}
        case 1n+after_second:
          first_line second_line = Mandelbrot.row(
            start_row, half_width, width, height, max_iterations
          ) Mandelbrot.row(
            1n+start_row, half_width, width, height, max_iterations
          )
          first_line <> second_line
            <> Mandelbrot.lines_remaining(
              after_second,
              2n+start_row,
              half_width,
              width,
              height,
              max_iterations
            )

def Mandelbrot.lines(
  height: Nat,
  width: Nat,
  half_width: Nat,
  max_iterations: Nat
) -> List<&2, String>:
  Mandelbrot.lines_remaining(
    height, 0n, half_width, width, height, max_iterations
  )

# --- Sierpinski ---

def Sierpinski.pick_character(in_set: Bool) -> Char:
  match in_set:
    case True{}:
      '*'
    case False{}:
      ' '

# The classic Sierpinski triangle is Pascal's triangle mod 2: cell
# (column, row) is filled exactly when column is a bitwise submask of row.
# That is one AND and one EQ per cell, which is why this picture does not
# need a parallel let.
def Sierpinski.character(column_index: Nat, actual_row: Nat) -> Char:
  Sierpinski.pick_character(
    U32.is_eq(
      U32.and(U32.from_nat(column_index), U32.from_nat(actual_row)),
      U32.from_nat(column_index)
    )
  )

def Sierpinski.row_segment(
  current_column: Nat,
  remaining_columns: Nat,
  actual_row: Nat
) -> String:
  match remaining_columns:
    case 0n:
      SNil{}
    case 1n+rest:
      SCon{
        Sierpinski.character(current_column, actual_row),
        Sierpinski.row_segment(1n+current_column, rest, actual_row)
      }

def Sierpinski.row(actual_row: Nat, width: Nat) -> String:
  Sierpinski.row_segment(0n, width, actual_row)

# Rows are drawn from the widest (height - 1) down to the narrowest (0), so
# the triangle sits point-up on the terminal. Nat.sub clamps at zero, which
# is why no guard is needed on the last step.
def Sierpinski.lines_remaining(
  remaining_rows: Nat,
  current_actual_row: Nat,
  width: Nat
) -> List<&2, String>:
  match remaining_rows:
    case 0n:
      Nil{}
    case 1n+rest:
      Sierpinski.row(current_actual_row, width)
        <> Sierpinski.lines_remaining(
          rest, Nat.sub(current_actual_row, 1n), width
        )

def Sierpinski.lines(height: Nat, width: Nat) -> List<&2, String>:
  Sierpinski.lines_remaining(height, Nat.sub(height, 1n), width)

# --- Main ---

def main() -> IO(Unit):

    IO.print("")

# LAWS.bend
# Claims about playground.bend. Each law is an open claim; PROOF.bend fills
# it in with a def of the same name.

import Base
import ./playground.bend as Playground

# The length of a repeated character run must equal the requested count.
# This is the only law that exercises a real induction; it mirrors the
# add_zero example from the guide.
law repeat_character_length:
  for count: Nat
  for character: Char
  {String.length(Playground.repeat_character(count, character)) == count : Nat}

# A point that has already escaped costs no further iterations, whatever the
# remaining budget is. Stated separately from the zero-budget law because
# both are needed to convince the checker that the escape loop short-circuits.
law mandelbrot_escaped_returns_zero:
  for remaining_iterations: Nat
  for real_part: F32
  for imaginary_part: F32
  for real_constant: F32
  for imaginary_constant: F32
  {Playground.Mandelbrot.escape_iterations(
    remaining_iterations,
    real_part,
    imaginary_part,
    real_constant,
    imaginary_constant,
    True{}
  ) == 0n : Nat}

# A zero budget returns zero regardless of the escape flag. Without this the
# checker would have to reason about the F32 constants, which Base does not
# support.
law mandelbrot_zero_iterations_returns_zero:
  for real_part: F32
  for imaginary_part: F32
  for real_constant: F32
  for imaginary_constant: F32
  for point_has_escaped: Bool
  {Playground.Mandelbrot.escape_iterations(
    0n,
    real_part,
    imaginary_part,
    real_constant,
    imaginary_constant,
    point_has_escaped
  ) == 0n : Nat}

# Induction on remaining_rows. The nested match on after_first is forced by
# the function's own shape: it distinguishes the one-row tail from the
# two-rows-plus-rest case. In the two-rows case the goal unfolds to
# 1n + 1n + length of the recursive list, and the hypothesis rewrites the

def Laws.mandelbrot_lines_remaining_length(
  remaining_rows,
  start_row,
  half_width,
  width,
  height,
  max_iterations
):

    case 0n:
      {==}
    case 1n+after_first:
      match after_first:
        case 0n:
          {==}

            max_iterations

What I have verified by reading, and what I am inferring
Verified by reading the guide and Base:

The names String.length, List.length, Nat.is_eq, Nat.sub, U32.is_eq, U32.and, U32.from_nat, F32.add/sub/mul/from_nat/is_gt, Bool, Char, F32, SCon, SNil, Nil{}, Con{}, IO.print, IO.pure, IO.bind all exist with the shapes I used.

The %e : P; e2 rewrite form and the {==} reflexivity term.

The rule that match may inspect a parameter or a pattern-bound variable, never a computed value.

The rule that mutual recursion is forbidden, and that termination reads arguments left to right.

The rule that do M<xs.., R>: desugars to M.bind and M.pure.

Inferring, not verified:

That {==} unfolds a function call whose arguments are all closed enough to reduce (for mandelbrot_escaped_returns_zero and the 0n cases).

That %e : P unifies P against the goal after the checker has normalized the goal's left-hand side.

That the termination checker accepts the nested match after_first: as a strict subterm decomposition.

That Mandelbrot.escape_iterations's + markers on real_part/imaginary_part/real_constant/imaginary_constant satisfy the affine checker, since each is used more than once.

That a parallel let inside a case branch is accepted.

That a pure let with a type annotation (x : T = v) is accepted inside a do block.

That SCon{character, ...} where character is + in the signature is accepted as a single use of character plus a separate use in the recursive call.

Ranked list of what I expect to fail, most likely first
Laws.mandelbrot_lines_remaining_length step. The % rewrite has to unify a P whose left side already shows 1n+1n+List.length(...) against a goal whose left side is still List.length(Mandelbrot.lines_remaining(1n+1n+after_second, ...)). If the checker does not normalise the goal's left side before unification, this law fails to typecheck. This is the single riskiest line in the whole deliverable.

Laws.repeat_character_length step. Same unification question, smaller term. If it fails, it fails on the same principle.

+ markers on Mandelbrot.escape_iterations. If the affine checker wants +new_real_part and +new_imaginary_part declared differently, or if 2.0/4.0 need : F32 annotations, the def fails before the laws ever run.

0.0 in Mandelbrot.character_at. The checker may not infer F32 for a bare decimal in argument position.

Laws.mandelbrot_zero_iterations_returns_zero. The {==} in the False{} branch relies on the inner match on remaining_iterations = 0n reducing inside the goal. If the checker does not reduce nested matches in {==}, both branches fail.

print_lines lambda binder. If ignored_result is flagged as unused or if the IO.bind quantity arguments are wrong, main fails.

do block with a bare IO expression as the last statement. If the last statement must be return ..., the desugaring fails.

U32.is_eq and U32.and in Sierpinski.character. These are used but not proven about; if Base's U32.and is a law rather than a def, nothing here depends on it, so this is only a risk if the checker needs the value to reduce somewhere. It does not, for the laws I stated.

Nat.sub in Sierpinski.lines_remaining. Termination only, not evaluation; if the checker treats Nat.sub as a non-decreasing function it may reject the recursive call. This is the second-riskiest structural point.

F32 literals -2.0, 2.5, 1.25. If Bend's lexer wants a digit before the dot, -2.0 parses but .5 would not; I used 0.5-style forms throughout, so this is mostly a formatting worry.
